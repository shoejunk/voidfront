cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(VoidfrontSimulation LANGUAGES CXX)
  enable_testing()
endif()
add_library(voidfront_sim STATIC voidfront_sim.cpp)
target_include_directories(voidfront_sim PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(voidfront_sim PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(voidfront_sim PRIVATE /W4 /permissive-)
endif()
add_executable(voidfront_sim_tests tests.cpp)
target_link_libraries(voidfront_sim_tests PRIVATE voidfront_sim)
add_test(NAME voidfront_sim_tests COMMAND voidfront_sim_tests)
add_executable(voidfront_headless headless.cpp)
target_link_libraries(voidfront_headless PRIVATE voidfront_sim)
