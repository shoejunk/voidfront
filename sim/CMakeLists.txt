cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(VoidfrontSimulation LANGUAGES CXX)
  enable_testing()
endif()
add_library(voidfront_sim STATIC voidfront_sim.cpp lockstep.cpp navigation.cpp)
# Content compatibility changes automatically when authoritative source changes.
# Normalize line endings so a checkout's CRLF policy cannot change the identity.
set(content_source "Voidfront authoritative content v1\n")
foreach(source IN ITEMS voidfront_sim.hpp voidfront_sim.cpp lockstep.hpp lockstep.cpp navigation.hpp navigation.cpp)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/${source}" content)
  string(REPLACE "\r\n" "\n" content "${content}")
  string(APPEND content_source "${source}\n${content}\n")
endforeach()
string(SHA256 content_digest "${content_source}")
string(SUBSTRING "${content_digest}" 0 16 content_id)
message(STATUS "Voidfront content SHA256: ${content_digest}")
target_compile_definitions(voidfront_sim PUBLIC VF_CONTENT_ID=0x${content_id}ULL)
target_include_directories(voidfront_sim PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(voidfront_sim PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(voidfront_sim PRIVATE /W4 /permissive-)
endif()
add_executable(voidfront_sim_tests tests.cpp)
target_link_libraries(voidfront_sim_tests PRIVATE voidfront_sim)
add_test(NAME voidfront_sim_tests COMMAND voidfront_sim_tests)
add_executable(voidfront_lockstep_tests lockstep_tests.cpp)
target_link_libraries(voidfront_lockstep_tests PRIVATE voidfront_sim)
add_test(NAME voidfront_lockstep_tests COMMAND voidfront_lockstep_tests)
add_executable(voidfront_headless headless.cpp)
target_link_libraries(voidfront_headless PRIVATE voidfront_sim)

add_executable(voidfront_navigation_tests navigation_tests.cpp)
target_link_libraries(voidfront_navigation_tests PRIVATE voidfront_sim)
add_test(NAME voidfront_navigation_tests COMMAND voidfront_navigation_tests)
add_executable(voidfront_navigation_probe navigation_probe.cpp)
target_link_libraries(voidfront_navigation_probe PRIVATE voidfront_sim)
