cmake_minimum_required(VERSION 3.24)
project(Voidfront VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>" CACHE STRING "Static MSVC runtime")
option(VOIDFRONT_BUILD_CLIENT "Build the Godot bridge" ON)
include(CTest)
add_subdirectory(sim)
if(WIN32)
  add_library(voidfront_session STATIC net/session.cpp)
  target_link_libraries(voidfront_session PUBLIC voidfront_sim ws2_32)
  target_include_directories(voidfront_session PUBLIC net)
  add_executable(voidfront_peer net/peer.cpp)
  target_link_libraries(voidfront_peer PRIVATE voidfront_session winmm)
  if(BUILD_TESTING)
    add_executable(voidfront_session_tests net/session_tests.cpp)
    target_link_libraries(voidfront_session_tests PRIVATE voidfront_session)
    add_test(NAME voidfront_session_tests COMMAND voidfront_session_tests)
    set_tests_properties(voidfront_session_tests PROPERTIES TIMEOUT 30)
  endif()
  if(MSVC)
    target_compile_options(voidfront_session PRIVATE /W4 /permissive-)
    target_compile_options(voidfront_peer PRIVATE /W4 /permissive-)
  endif()
endif()
if(VOIDFRONT_BUILD_CLIENT)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/godot-cpp/CMakeLists.txt")
    message(FATAL_ERROR "Run tools/bootstrap.ps1 to fetch pinned godot-cpp")
  endif()
  set(GODOTCPP_API_VERSION "4.7" CACHE STRING "Pinned compatible Godot API")
  set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/extension/build_profile.json" CACHE PATH "Bridge API profile")
  set(GODOTCPP_DISABLE_EXCEPTIONS OFF CACHE BOOL "Use standard C++ exception support")
  add_subdirectory(third_party/godot-cpp)
  add_library(voidfront_bridge SHARED extension/bridge.cpp)
  target_link_libraries(voidfront_bridge PRIVATE voidfront_sim godot-cpp)
  target_include_directories(voidfront_bridge PRIVATE sim)
  set_target_properties(voidfront_bridge PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/client/bin/$<CONFIG>"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/client/bin/$<CONFIG>"
    OUTPUT_NAME "voidfront_bridge")
endif()
